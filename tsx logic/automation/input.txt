import React from 'react';

import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene01PapaFormula',
  durationInSeconds: 20,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene01PapaFormula: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Camera: slow push-in
  const camScale = interpolate(frame, [0, durationInFrames], [1, 1.09], { extrapolateRight: 'clamp' });

  // Screen shake on impact word "गरीब" (frame 210)
  const shakeAmt = interpolate(frame, [210, 216, 228], [0, 14, 0], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });
  const shakeX = Math.sin(frame * 2.3) * shakeAmt;
  const shakeY = Math.cos(frame * 1.9) * shakeAmt;

  // Coin drain animation (micro)
  const coins = new Array(6).fill(0).map((_, i) => {
    const start = 220 + i * 12;
    const fall = interpolate(frame, [start, start + 50], [0, 340], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.in(Easing.quad) });
    const op = interpolate(frame, [start, start + 10, start + 40, start + 50], [0, 1, 1, 0], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });
    return { fall, op, x: 1490 + seededRandom(i + 7) * 160 };
  });

  return (
    <AbsoluteFill style={{ backgroundColor: '#84934A', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient + Shape Layers + Texture + Particles + Content + Vignette + Watermark */}
      <AbsoluteFill style={{ transform: `scale(${camScale}) translate(${shakeX}px, ${shakeY}px)` }}>

        {/* Dynamic gradient background */}
        <AbsoluteFill style={{ background: 'radial-gradient(ellipse at 30% 20%, #96A559 0%, #84934A 45%, #5C6635 100%)' }} />

        {/* Punk collage torn strips */}
        <div style={{ position: 'absolute', top: -60, left: -100, width: 2200, height: 150, background: '#000000', transform: `rotate(-3deg) translateX(${interpolate(frame, [0, 30], [-300, 0], { extrapolateRight: 'clamp', easing: Easing.out(Easing.cubic) })}px)`, opacity: 0.9 }} />
        <div style={{ position: 'absolute', bottom: -70, left: -100, width: 2200, height: 170, background: '#ECECEC', transform: `rotate(2.2deg) translateX(${interpolate(frame, [5, 35], [300, 0], { extrapolateRight: 'clamp', easing: Easing.out(Easing.cubic) })}px)`, clipPath: 'polygon(0 18%, 4% 8%, 9% 16%, 15% 6%, 22% 14%, 30% 5%, 38% 15%, 47% 7%, 56% 13%, 64% 4%, 73% 12%, 82% 6%, 90% 14%, 100% 8%, 100% 100%, 0 100%)', opacity: 0.95 }} />

        {/* Animated transparent shape layers */}
        <div style={{ position: 'absolute', top: 180, right: -160, width: 560, height: 560, borderRadius: '50%', border: '3px solid #ECECEC', opacity: 0.07, transform: `rotate(${frame * 0.12}deg) scale(${1 + Math.sin(frame * 0.03) * 0.04})` }} />
        <div style={{ position: 'absolute', bottom: 120, left: -120, width: 420, height: 420, background: '#000000', opacity: 0.05, transform: `rotate(${18 + frame * 0.06}deg)` }} />

        {/* Motion texture: halftone dots */}
        {new Array(28).fill(0).map((_, i) => (
          <div key={`d${i}`} style={{ position: 'absolute', left: 100 + (i % 7) * 60, top: 760 + Math.floor(i / 7) * 52, width: 10 + (i % 3) * 4, height: 10 + (i % 3) * 4, borderRadius: '50%', background: '#000000', opacity: 0.12 }} />
        ))}

        {/* Floating particles */}
        {new Array(14).fill(0).map((_, i) => {
          const px = seededRandom(i * 3 + 1) * 1920;
          const py = (seededRandom(i * 5 + 2) * 1080 + frame * (0.3 + seededRandom(i) * 0.5)) % 1080;
          return <div key={`p${i}`} style={{ position: 'absolute', left: px, top: py, width: 5, height: 5, background: '#F4F2F2', opacity: 0.25, transform: `rotate(${frame + i * 30}deg)` }} />;
