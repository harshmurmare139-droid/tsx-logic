import React from 'react';

import {
  useCurrentFrame,
  useVideoConfig,
  interpolate,
  spring,
  AbsoluteFill,
  Easing,
} from 'remotion';

export const compositionConfig = {
  id: 'Scene11_LoopholeTeaser',
  durationInSeconds: 10,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene11_LoopholeTeaser: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({
      frame: Math.max(0, frame - delay),
      fps,
      config: { damping: 12, mass: 0.4, stiffness: 120, ...config },
    });

  const exitOp = interpolate(
    frame,
    [durationInFrames - 15, durationInFrames],
    [1, 0],
    { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' }
  );

  // Background gradient shift
  const bgHue = interpolate(frame, [0, durationInFrames], [0, 15], {
    extrapolateRight: 'clamp',
  });

  // Shape layers — floating geometric forms
  const shapes = Array.from({ length: 6 }, (_, i) => {
    const seed = seededRandom(i + 11);
    const size = 120 + seed * 300;
    const startX = seed * 1920;
    const startY = seededRandom(i + 50) * 1080;
    const drift = interpolate(frame, [0, durationInFrames], [0, 40 + seed * 60]);
    const rotation = interpolate(frame, [0, durationInFrames], [0, 20 + seed * 30]);
    const scaleBreath = 1 + Math.sin(frame * 0.03 + i) * 0.05;
    const shapeType = i % 3;
    return (
      <div
        key={i}
        style={{
          position: 'absolute',
          left: startX + drift * (i % 2 === 0 ? 1 : -1),
          top: startY + Math.sin(frame * 0.02 + i) * 20,
          width: size,
          height: size,
          borderRadius: shapeType === 0 ? '50%' : shapeType === 1 ? '12px' : '0px',
          border: `2px solid rgba(232, 237, 242, ${0.04 + seed * 0.04})`,
          background: `rgba(250, 185, 91, ${0.03 + seed * 0.03})`,
          transform: `rotate(${rotation + (shapeType === 2 ? 45 : 0)}deg) scale(${scaleBreath})`,
        }}
      />
    );
  });

  // Motion texture — diagonal lines
  const lines = Array.from({ length: 12 }, (_, i) => {
    const yPos = i * 100 - 50;
    const drift = interpolate(frame, [0, durationInFrames], [0, 80]);
    return (
      <div
        key={`line-${i}`}
        style={{
          position: 'absolute',
          left: -200 + drift,
          top: yPos,
          width: 2400,
          height: 1,
          background: `rgba(232, 237, 242, ${0.03 + (i % 3) * 0.01})`,
          transform: 'rotate(-25deg)',
        }}
      />
    );
  });

  // Floating particles
  const particles = Array.from({ length: 20 }, (_, i) => {
    const seed = seededRandom(i + 100);
    const x = seed * 1920;
    const y = seededRandom(i + 200) * 1080;
    const particleSize = 2 + seed * 4;
    const floatY = Math.sin(frame * 0.04 + i * 2) * 15;
    const floatX = Math.cos(frame * 0.03 + i) * 10;
    const op = interpolate(
      Math.sin(frame * 0.05 + i),
      [-1, 1],
      [0.1, 0.5]
    );
    return (
      <div
        key={`p-${i}`}
        style={{
          position: 'absolute',
          left: x + floatX,
          top: y + floatY,
          width: particleSize,
          height: particleSize,
          borderRadius: '50%',
          background: i % 3 === 0 ? '#FAB95B' : '#E8EDF2',
          opacity: op,
        }}
      />
    );
  });

  // Camera zoom effect
  const cameraScale = interpolate(frame, [0, durationInFrames], [1.02, 1], {
    extrapolateRight: 'clamp',
  });
  const cameraPanX = interpolate(frame, [0, durationInFrames], [-10, 5]);

  // Main content — lock icon + text reveal
  const lockScale = sp(10, { damping: 10, stiffness: 100 });
  const titleOp = sp(20);
  const titleY = interpolate(sp(20), [0, 1], [60, 0]);
  const subtitleOp = sp(35);
  const subtitleY = interpolate(sp(35), [0, 1], [40, 0]);
  const badgeOp = sp(50);
  const badgeScale = sp(50, { damping: 8, stiffness: 150 });

  // Pulsing glow on lock
  const glowPulse = Math.sin(frame * 0.08) * 0.3 + 0.7;

  // Screen shake at frame 15
  const shakeX = frame >= 14 && frame <= 20 ? Math.sin(frame * 8) * 3 : 0;
  const shakeY = frame >= 14 && frame <= 20 ? Math.cos(frame * 8) * 2 : 0;

  // Vignette
  const vignette = (
    <div
      style={{
        position: 'absolute',
        inset: 0,
        background:
          'radial-gradient(ellipse at center, transparent 50%, rgba(20,20,40,0.6) 100%)',
        pointerEvents: 'none',
      }}
    />
  );

  // Watermark
  const watermarkOp = interpolate(frame, [0, 30], [0, 0.3], {
    extrapolateRight: 'clamp',
  });

  return (
    <AbsoluteFill
      style={{
        backgroundColor: '#4647AE',
        overflow: 'hidden',
        opacity: exitOp,
      }}
    >
      {/* Dynamic gradient background */}
      <AbsoluteFill
        style={{
          background: `linear-gradient(${135 + bgHue}deg, #4647AE 0%, #363799 40%, #2d2e7a 70%, #3a3ba0 100%)`,
          transform: `scale(${cameraScale}) translateX(${cameraPanX + shakeX}px) translateY(${shakeY}px)`,
        }}
      >
        {/* Shape layers */}
        {shapes}

        {/* Motion texture lines */}
        {lines}

        {/* Particles */}
        {particles}

        {/* Main Content */}
        <AbsoluteFill
          style={{
            display: 'flex',
            flexDirection: 'column',
            alignItems: 'center',
            justifyContent: 'center',
          }}
        >
          {/* Lock icon SVG */}
          <div
            style={{
              transform: `scale(${lockScale})`,
              marginBottom: 30,
              filter: `drop-shadow(0 0 ${20 * glowPulse}px rgba(250,185,91,${0.5 * glowPulse}))`,
            }}
          >
            <svg
              width="110"
              height="130"
              viewBox="0 0 110 130"
              fill="none"
              xmlns="[w3.org](http://www.w3.org/2000/svg)"
            >
              <rect
                x="10"
                y="55"
                width="90"
                height="70"
                rx="12"
                fill="#FAB95B"
              />
              <path
                d="M30 55V40C30 26.2 41.2 15 55 15C68.8 15 80 26.2 80 40V55"
                stroke="#E8EDF2"
                strokeWidth="8"
                strokeLinecap="round"
                fill="none"
              />
              <circle cx="55" cy="88" r="10" fill="#4647AE" />
              <rect x="52" y="93" width="6" height="15" rx="3" fill="#4647AE" />
            </svg>
          </div>

          {/* Title */}
          <div
            style={{
              fontFamily: "'Inter', 'Poppins', sans-serif",
              fontSize: 72,
              fontWeight: 800,
              color: '#E8EDF2',
              textAlign: 'center',
              opacity: titleOp,
              transform: `translateY(${titleY}px)`,
              letterSpacing: '-1px',
              lineHeight: 1.1,
            }}
          >
            PENALTY-FREE SWITCH
          </div>

          {/* Subtitle */}
          <div
            style={{
              fontFamily: "'Noto Sans Devanagari', sans-serif",
              fontSize: 38,
              fontWeight: 600,
              color: '#FAB95B',
              textAlign: 'center',
              opacity: subtitleOp,
              transform: `translateY(${subtitleY}px)`,
              marginTop: 18,
              letterSpacing: '3px',

          >
            THE ULTIMATE LOOPHOLE
          </div>

          {/* Badge */}
          <div

              border: '2px solid rgba(250,185,91,0.4)',
