import React from 'react';

import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene1Antar',
  durationInSeconds: 12,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene1Antar: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  const labelOp = sp(10);
  const numScale = sp(25);
  const titleScale = sp(50);
  const underlineW = interpolate(sp(80), [0, 1], [0, 700]);
  const subOp = sp(110);

  // Camera slow zoom
  const camScale = interpolate(frame, [0, durationInFrames], [1, 1.05]);

  return (
    <AbsoluteFill style={{ backgroundColor: '#FFDE42', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient overlay */}
      <AbsoluteFill style={{ background: 'radial-gradient(circle at 30% 40%, rgba(223,241,241,0.4) 0%, transparent 60%)' }} />

      {/* Shape layers - drifting */}
      {[0, 1, 2, 3, 4].map(i => {
        const r = seededRandom(i + 1);
        const drift = Math.sin(frame * 0.01 + i) * 30;
        return (
          <div key={i} style={{
            position: 'absolute',
            left: `${r * 80}%`,
            top: `${seededRandom(i + 7) * 80}%`,
            width: 200 + r * 180,
            height: 200 + r * 180,
            borderRadius: i % 2 === 0 ? '50%' : '20%',
            border: '3px solid #1F150C',
            opacity: 0.05,
            transform: `translateY(${drift}px) rotate(${frame * 0.1 + i * 30}deg)`,
          }} />
        );
      })}

      {/* Grid texture */}
      <AbsoluteFill style={{
        backgroundImage: 'linear-gradient(#1F150C 1px, transparent 1px), linear-gradient(90deg, #1F150C 1px, transparent 1px)',
        backgroundSize: '80px 80px',
        opacity: 0.04,
      }} />

      {/* Floating particles */}
      {Array.from({ length: 18 }).map((_, i) => {
        const r = seededRandom(i + 100);
        const y = (seededRandom(i + 200) * 1080 + frame * (0.5 + r)) % 1080;
        return (
          <div key={`p${i}`} style={{
            position: 'absolute',
            left: `${r * 100}%`,
            top: y,
            width: 6,
            height: 6,
            backgroundColor: '#080616',
            borderRadius: '50%',
            opacity: 0.3,
          }} />
        );
      })}

      {/* Main content centered */}
      <AbsoluteFill style={{
        display: 'flex',
        alignItems: 'center',
        justifyContent: 'center',
        flexDirection: 'column',
        transform: `scale(${camScale})`,
      }}>
        {/* Top label */}
        <div style={{
          fontFamily: 'Inter, Poppins',
          fontSize: 38,
          fontWeight: 700,
          letterSpacing: 8,
          color: '#080616',
          opacity: labelOp,
          padding: '12px 32px',
          border: '3px solid #080616',
          marginBottom: 40,

          FUNDAMENTAL
