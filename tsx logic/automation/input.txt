import React from 'react';

import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene1_QuestionOpening',
  durationInSeconds: 10,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene1_QuestionOpening: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Camera zoom in slowly
  const cameraZoom = interpolate(frame, [0, durationInFrames], [1, 1.08], { extrapolateRight: 'clamp' });

  // Question mark draw
  const qDraw = interpolate(sp(20), [0, 1], [0, 1]);
  const qScale = sp(20);

  // Word reveals
  const word1 = sp(45);
  const word2 = sp(70);
  const word3 = sp(95);
  const lineExpand = interpolate(sp(120), [0, 1], [0, 1]);
  const brandReveal = sp(150);

  // Floating particles
  const particles = Array.from({ length: 30 }).map((_, i) => {
    const seed = i + 1;
    const x = seededRandom(seed) * 1920;
    const y = seededRandom(seed * 2) * 1080;
    const drift = Math.sin(frame * 0.02 + i) * 15;
    const size = 3 + seededRandom(seed * 3) * 5;
    return { x, y: y + drift, size, opacity: 0.15 + seededRandom(seed * 4) * 0.2 };
  });

  return (
    <AbsoluteFill style={{ backgroundColor: '#DFF1F1', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient overlay */}
      <AbsoluteFill style={{
        background: 'radial-gradient(circle at 50% 40%, rgba(28,28,28,0.08) 0%, rgba(223,241,241,0) 70%)',
      }} />

      {/* Grid texture */}
      <svg width="1920" height="1080" style={{ position: 'absolute', opacity: 0.08 }}>
        <defs>
          <pattern id="grid1" width="80" height="80" patternUnits="userSpaceOnUse">
            <path d="M 80 0 L 0 0 0 80" fill="none" stroke="#1C1C1C" strokeWidth="1" />
          </pattern>
        </defs>
        <rect width="1920" height="1080" fill="url(#grid1)" />
      </svg>

      {/* Shape layers */}
      <div style={{
        position: 'absolute', top: 100, left: -150,
        width: 600, height: 600, borderRadius: '50%',
        border: '2px solid #1C1C1C', opacity: 0.05,
        transform: `rotate(${frame * 0.2}deg) scale(${1 + Math.sin(frame * 0.02) * 0.05})`,
      }} />
      <div style={{
        position: 'absolute', bottom: -200, right: -100,
        width: 700, height: 700,
        border: '2px solid #FFC107', opacity: 0.08,
        transform: `rotate(${-frame * 0.15}deg)`,
      }} />
      <div style={{
        position: 'absolute', top: '30%', right: '15%',
        width: 250, height: 250,
        background: '#e9d700', opacity: 0.06,
        transform: `rotate(45deg) scale(${1 + Math.sin(frame * 0.03) * 0.08})`,
      }} />

      {/* Particles */}
      {particles.map((p, i) => (
        <div key={i} style={{
          position: 'absolute', left: p.x, top: p.y,
          width: p.size, height: p.size, borderRadius: '50%',
          background: '#1C1C1C', opacity: p.opacity,
        }} />
      ))}

      <AbsoluteFill style={{ transform: `scale(${cameraZoom})` }}>
        {/* Giant Question Mark */}
        <div style={{
          position: 'absolute', top: '8%', left: '50%',
          transform: `translateX(-50%) scale(${qScale})`,
          opacity: qDraw,
        }}>
          <svg width="320" height="380" viewBox="0 0 320 380">
            <path
              d="M 60 110 Q 60 40 160 40 Q 260 40 260 120 Q 260 180 180 210 Q 160 220 160 260"
              fill="none" stroke="#1C1C1C" strokeWidth="22" strokeLinecap="round"
              strokeDasharray="800"
              strokeDashoffset={800 - qDraw * 800}
            />
            <circle cx="160" cy="330" r="18" fill="#ff0000" opacity={qDraw} />
          </svg>
        </div>

        {/* Headline stacked words */}
        <div style={{
          position: 'absolute', top: '52%', left: 0, right: 0,
          textAlign: 'center', fontFamily: 'Yantra One',
        }}>
          <div style={{
            fontSize: 102, color: '#1C1C1C', lineHeight: 1.1,
            opacity: word1, transform: `translateY(${(1 - word1) * 40}px)`,
            letterSpacing: -2,
          }}>
            यही वो <span style={{ color: '#ff0000' }}>सवाल</span> है
          </div>
          <div style={{
            fontSize: 76, color: '#2D2D2D', lineHeight: 1.2, marginTop: 20,
            opacity: word2, transform: `translateY(${(1 - word2) * 30}px)`,
            fontFamily: 'Noto Sans Devanagari', fontWeight: 700,
          }}>
            जो आज हम खोलने वाले हैं
          </div>
        </div>

        {/* Underline expand */}
        <div style={{
          position: 'absolute', top: '78%', left: '50%',
          transform: `translateX(-50%) scaleX(${lineExpand})`,
          width: 600, height: 6, background: '#FFC107',
          transformOrigin: 'center',
        }} />

        {/* Brand reveal */}
        <div style={{
          position: 'absolute', top: '82%', left: 0, right: 0,
          textAlign: 'center',
          opacity: brandReveal, transform: `translateY(${(1 - brandReveal) * 20}px)`,

          <div style={{
            display: 'inline-block', padding: '12px 36px',
            background: '#1C1C1C', color: '#e9d700',
            fontFamily: 'Inter', fontWeight: 800, fontSize: 38,
            letterSpacing: 3,
          }}>
            PROBUFFET HQ

        background: 'radial-gradient(ellipse at center, transparent 50%, rgba(28,28,28,0.25) 100%)',
        pointerEvents: 'none',
      }} />

      {/* Watermark */}
      <div style={{
        position: 'absolute', bottom: 30, right: 45,
        fontFamily: 'Inter', fontSize: 22, fontWeight: 600,
        color: '#9E9E9E', opacity: 0.3,
      }}>
        ProBuffet HQ
      </div>
    </AbsoluteFill>
  );
};

export default Scene1_QuestionOpening;

// Gear helper component
const Gear: React.FC<{ radius: number; teeth: number; color: string }> = ({ radius, teeth, color }) => {
  const toothHeight = radius * 0.2;
  const points: string[] = [];
  for (let i = 0; i < teeth * 2; i++) {
    const angle = (i / (teeth * 2)) * Math.PI * 2;
    const r = i % 2 === 0 ? radius + toothHeight : radius;
    points.push(`${Math.cos(angle) * r},${Math.sin(angle) * r}`);
  }
  return <polygon points={points.join(' ')} fill={color} />;
};
