  fps: 30,
  width: 1920,
  height: 1080,

  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

  const shape2Rot = -frame * 0.1;

      />

        style={{
          position: 'absolute', right: 200, top: 100, width: 0, height: 0,
          borderLeft: '180px solid transparent', borderRight: '180px solid transparent',
          borderBottom: '300px solid #609966', opacity: 0.05,
          transform: `translateY(${shape3Y}px) rotate(${frame * 0.08}deg)`,
        }}
      />

      {/* Motion texture — diagonal lines grid */}
      <svg width="1920" height="1080" style={{ position: 'absolute', inset: 0, opacity: 0.07 }}>
        <defs>
          <pattern id="lines11" width="40" height="40" patternUnits="userSpaceOnUse" patternTransform={`translate(${(frame * 0.5) % 40}, 0)`}>

            borderRadius: '50%', backgroundColor: '#93B5C6', opacity: p.op,
            boxShadow: '0 0 8px rgba(147,181,198,0.6)',
          }}
        />

      {/* Main content wrapper with camera zoom */}
