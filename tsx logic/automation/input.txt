import React from 'react';

import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene01_MonsoonAlert',
  durationInSeconds: 11.9,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene01_MonsoonAlert: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // camera slow zoom
  const camZoom = interpolate(frame, [0, durationInFrames], [1, 1.06], { extrapolateRight: 'clamp' });

  // cloud drift
  const cloudX = interpolate(frame, [0, durationInFrames], [-40, 60]);

  // alert badge entrance
  const badge = sp(20);
  // headline reveal
  const head = sp(40);
  // sub reveal
  const sub = sp(70);
  // stat reveal
  const stat = sp(95);

  // blur/clarity metaphor — "धुंधली तस्वीर"
  const blurAmt = interpolate(frame, [40, 110], [14, 2], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });

  return (
    <AbsoluteFill style={{ backgroundColor: '#EAECF0', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient background layer */}
      <AbsoluteFill
        style={{
          background: 'radial-gradient(circle at 30% 20%, #F4F5F8 0%, #EAECF0 45%, #DCDEE6 100%)',
        }}
      />

      {/* Shape Layers — transparent drifting geometry */}
      <AbsoluteFill style={{ transform: `scale(${camZoom})` }}>
        <div style={{
          position: 'absolute', top: '12%', left: '8%', width: 520, height: 520,
          borderRadius: '50%', border: '2px solid #F62440', opacity: 0.05,
          transform: `rotate(${frame * 0.15}deg) scale(${1 + Math.sin(frame * 0.02) * 0.05})`,
        }} />
        <div style={{
          position: 'absolute', bottom: '10%', right: '6%', width: 460, height: 460,
          backgroundColor: '#000000', opacity: 0.04, borderRadius: 40,
          transform: `rotate(${-frame * 0.1}deg)`,
        }} />
      </AbsoluteFill>

      {/* Motion texture — rain lines */}
      <AbsoluteFill>
        {new Array(40).fill(0).map((_, i) => {
          const r = seededRandom(i + 1);
          const x = r * 1920;
          const speed = 4 + r * 6;
          const y = ((frame * speed) + r * 1080) % 1180 - 100;
          return (
            <div key={i} style={{
              position: 'absolute', left: x, top: y, width: 2, height: 28,
              background: 'linear-gradient(#9AA0B0, transparent)', opacity: 0.18,
            }} />
          );
        })}
      </AbsoluteFill>

      {/* Floating particles */}
      <AbsoluteFill>
        {new Array(18).fill(0).map((_, i) => {
          const r = seededRandom(i + 50);
          const x = r * 1920;
          const y = ((frame * (0.4 + r)) + r * 1080) % 1080;
          return (
            <div key={i} style={{
              position: 'absolute', left: x, top: y, width: 6, height: 6, borderRadius: '50%',
              backgroundColor: '#F62440', opacity: 0.12,
            }} />
          );
        })}
      </AbsoluteFill>

      {/* Drifting cloud SVG (storytelling element) */}
      <div style={{ position: 'absolute', top: 90, left: 760 + cloudX, opacity: 0.9 }}>
        <svg width="420" height="200" viewBox="0 0 420 200">
          <g fill="#C7CBD6" opacity="0.85">
            <ellipse cx="140" cy="120" rx="110" ry="60" />
            <ellipse cx="240" cy="100" rx="120" ry="70" />
            <ellipse cx="320" cy="130" rx="80" ry="48" />
          </g>
          <g stroke="#9AA0B0" strokeWidth="4" strokeLinecap="round" opacity="0.5">
            {new Array(6).fill(0).map((_, i) => (
              <line key={i} x1={120 + i * 35} y1={175}
                x2={112 + i * 35} y2={175 + 22 + Math.sin((frame + i * 10) * 0.2) * 6} />
            ))}
          </g>
        </svg>
      </div>

      {/* MAIN CONTENT */}
      <AbsoluteFill style={{ justifyContent: 'center', paddingLeft: 140, paddingRight: 140 }}>

        {/* Alert badge */}
        <div style={{
          opacity: badge, transform: `translateX(${(1 - badge) * -60}px)`,
          display: 'inline-flex', alignItems: 'center', gap: 14,
          backgroundColor: '#F62440', padding: '12px 26px', borderRadius: 50,
          alignSelf: 'flex-start', marginBottom: 34, boxShadow: '0 8px 30px rgba(246,36,64,0.3)',
        }}>
          <svg width="30" height="30" viewBox="0 0 24 24" fill="none">
            <path d="M12 2 L22 20 H2 Z" stroke="#FFFFFF" strokeWidth="2" strokeLinejoin="round" />

        {/* IMD stat callout card */}
        <div style={{
          opacity: stat, transform: `translateY(${(1 - stat) * 30}px) scale(${0.95 + stat * 0.05})`,
          marginTop: 44, display: 'flex', alignItems: 'center', gap: 26,
          backgroundColor: '#FFFFFF', borderLeft: '8px solid #F62440',
