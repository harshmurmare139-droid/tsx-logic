  width: 1920,
  height: 1080,
};

  return x - Math.floor(x);
};

          </pattern>
        </defs>
        <rect x="1250" y="0" width="670" height="1080" fill="url(#dots01)" transform={`translate(${Math.sin(frame * 0.01) * 12}, 0)`} />
        <rect x="0" y="760" width="600" height="320" fill="url(#dots01)" />
