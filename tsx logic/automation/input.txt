            position: 'absolute',
            left: x - size / 2,
            top: y - size / 2,
            width: size,
            height: size,

        );

      <svg width="1920" height="1080" style={{ position: 'absolute', opacity: 0.08 }}>
        <defs>
          <pattern id="grid11" width="80" height="80" patternUnits="userSpaceOnUse">
            <path d="M 80 0 L 0 0 0 80" fill="none" stroke="#1D2128" strokeWidth="1" />
          </pattern>
        </defs>
        <rect width="1920" height="1080" fill="url(#grid11)" transform={`translate(${frame * 0.3}, ${frame * 0.2})`} />
      </svg>

      {/* Floating particles */}
      {Array.from({ length: 30 }).map((_, i) => {
        const seed = seededRandom(i + 50);
        const px = (seededRandom(i + 60) * 1920 + frame * (0.3 + seed * 0.8)) % 1920;
        const py = (seededRandom(i + 70) * 1080 + frame * 0.4) % 1080;
        return (
          <div key={`p-${i}`} style={{
            position: 'absolute',
            left: px,
            top: py,

            height: 4 + seed * 6,

            fontFamily: 'Space Mono, monospace',
            fontSize: 28,

            LE PARADOXE FRANÇAIS
