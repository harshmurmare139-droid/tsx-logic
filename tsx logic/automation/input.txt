    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // 99 counter
  const counter = Math.floor(interpolate(frame, [10, 70], [0, 99], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' }));

  // Pie sweep — 99% red, 1% green
  const pieProgress = interpolate(frame, [20, 90], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });
  const angle99 = pieProgress * 356.4; // 99% of 360
  const cx = 1450, cy = 540, r = 240;
  const a1 = -90;
  const a2 = a1 + angle99;
  const x1 = cx + r * Math.cos((a1 * Math.PI) / 180);
  const y1 = cy + r * Math.sin((a1 * Math.PI) / 180);
  const x2 = cx + r * Math.cos((a2 * Math.PI) / 180);
  const y2 = cy + r * Math.sin((a2 * Math.PI) / 180);
  const largeArc = angle99 > 180 ? 1 : 0;
  const piePath = `M ${cx} ${cy} L ${x1} ${y1} A ${r} ${r} 0 ${largeArc} 1 ${x2} ${y2} Z`;

  // Camera
  const camScale = interpolate(frame, [0, durationInFrames], [1, 1.04]);

  return (
    <AbsoluteFill style={{ backgroundColor: '#0D0B61', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient */}
      <AbsoluteFill style={{
        background: 'radial-gradient(circle at 30% 50%, #1a1690 0%, #0D0B61 70%, #050332 100%)',
      }} />

      {/* Shapes */}
      <svg width="1920" height="1080" style={{ position: 'absolute' }}>
        <rect x="100" y="800" width="600" height="200" fill="#B4E50D" opacity="0.04"
          transform={`rotate(${-10 + Math.sin(frame * 0.02) * 3} 400 900)`} />
        <circle cx="200" cy="200" r="180" fill="none" stroke="#FFF9BD" strokeWidth="2" opacity="0.06" />
      </svg>

      {/* Hex grid texture */}
      <svg width="1920" height="1080" style={{ position: 'absolute', opacity: 0.05 }}>
        <defs>
          <pattern id="hex1" width="60" height="52" patternUnits="userSpaceOnUse">
            <polygon points="30,2 58,17 58,47 30,62 2,47 2,17" fill="none" stroke="#B4E50D" strokeWidth="0.6" />
          </pattern>
        </defs>
        <rect width="1920" height="1080" fill="url(#hex1)" />
      </svg>

      {/* Particles */}
      {Array.from({ length: 30 }).map((_, i) => {
        const px = (seededRandom(i) * 1920 + frame * 0.3) % 1920;
        const py = seededRandom(i + 200) * 1080;
        return (
          <div key={`p-${i}`} style={{
            position: 'absolute',
            left: px,
            top: py,
            width: 3,
            height: 3,
            background: '#FFF9BD',
            opacity: 0.4,
            borderRadius: '50%',
          }} />
        );
      })}

      <AbsoluteFill style={{ transform: `scale(${camScale})` }}>
        {/* LEFT — big stat */}
        <div style={{
          position: 'absolute',
          left: 100,
          top: 200,
          width: 800,
          opacity: sp(0),
        }}>
          <div style={{
            fontFamily: 'Inter',
            fontSize: 30,
            color: '#B4E50D',
            letterSpacing: 6,
            fontWeight: 700,
            marginBottom: 16,
          }}>
            ▎ HARSH STATISTIC
          </div>

          <div style={{
            fontFamily: 'Yantra One',
            fontSize: 360,
            color: '#FFF9BD',
            fontWeight: 900,
            lineHeight: 0.95,
            display: 'flex',
            alignItems: 'baseline',
            textShadow: '0 0 50px rgba(255,249,189,0.3)',
          }}>
            {counter}
            <span style={{
              fontSize: 180,
              color: '#B4E50D',
              marginLeft: 8,
            }}>
              %
            </span>
          </div>

          <div style={{
            fontFamily: 'Noto Sans Devanagari',
            fontSize: 56,
            color: '#FFF9BD',
            fontWeight: 700,
            marginTop: 30,
            opacity: sp(40),
          }}>
            भारतीय निवेशक
          </div>
          <div style={{
            fontFamily: 'Noto Sans Devanagari',
            fontSize: 44,
            color: '#FFF9BD',
            opacity: sp(60) * 0.8,
            marginTop: 12,
            fontWeight: 500,
          }}>
            कभी ये सीख ही नहीं पाते
          </div>
        </div>

        {/* RIGHT — Pie chart */}
        <svg width="600" height="600" style={{
          position: 'absolute',
          right: 100,
          top: 240,
          opacity: sp(20),
        }}>
          {/* Outer ring decor */}
          <circle cx="300" cy="300" r="285" fill="none" stroke="#B4E50D" strokeWidth="2" opacity="0.3" strokeDasharray="6 6"
            style={{ transformOrigin: '300px 300px', transform: `rotate(${frame}deg)` }} />

          {/* Background full circle (1% piece — green) */}
          <circle cx="300" cy="300" r="240" fill="#B4E50D" />

          {/* 99% red sweep */}
          <g transform="translate(-1150, -240)">
            <path d={piePath} fill="#ff0000" opacity="0.9" />

          {/* Center hole */}
          <circle cx="300" cy="300" r="120" fill="#0D0B61" stroke="#FFF9BD" strokeWidth="3" />

        <div style={{
