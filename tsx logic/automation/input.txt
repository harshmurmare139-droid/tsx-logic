import React from 'react';
import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene01_SalaireMoyen',
  durationInSeconds: 17,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene01_SalaireMoyen: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Number counter animation
  const counter = interpolate(frame, [20, 80], [0, 2493], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.out(Easing.cubic),
  });

  // Camera zoom
  const camScale = interpolate(frame, [0, durationInFrames], [1, 1.08]);

  // Sub reveal
  const subOp = sp(90);
  const barW = interpolate(frame, [110, 200], [0, 900], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.out(Easing.cubic),
  });

  const lossOp = sp(210);
  const flashY = Math.sin(frame * 0.1) * 3;

  return (
    <AbsoluteFill style={{ backgroundColor: '#99C2FF', overflow: 'hidden' }}>
      {/* Gradient overlay */}
      <AbsoluteFill style={{ background: 'radial-gradient(circle at 30% 30%, rgba(255,255,255,0.35) 0%, rgba(153,194,255,0) 60%)' }} />

      {/* Shape Layers */}
      <div style={{ position: 'absolute', top: -200, right: -150, width: 700, height: 700, borderRadius: '50%', background: '#FE7F2D', opacity: 0.06, transform: `scale(${1 + Math.sin(frame * 0.02) * 0.05})` }} />

        <defs>
          <pattern id="grid1" width="80" height="80" patternUnits="userSpaceOnUse">

export const compositionConfig = {
  id: 'Scene05_MoinsZeroSept',
  durationInSeconds: 17,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene05_MoinsZeroSept: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Number appears with impact
  const numScale = sp(90, { stiffness: 150, damping: 10 });
  const impactShake = frame > 90 && frame < 120 ? Math.sin(frame * 2) * 6 : 0;

  const labelOp = sp(160);
  const bottomOp = sp(220);

  // Downward arrow animation
  const arrowY = interpolate(sp(70), [0, 1], [-100, 0]);

  return (
    <AbsoluteFill style={{ backgroundColor: '#99C2FF', overflow: 'hidden' }}>
      {/* Gradient */}
      <AbsoluteFill style={{ background: 'radial-gradient(circle at 50% 50%, rgba(254,127,45,0.18) 0%, rgba(153,194,255,0) 55%)' }} />

          UN CHIFFRE FAUSSEMENT ANODIN
        </div>

        {/* Down arrow */}

          <svg width="80" height="80" viewBox="0 0 100 100">
            <path d="M50 10 L50 80 M25 60 L50 85 L75 60" fill="none" stroke="#FE7F2D" strokeWidth="8" strokeLinecap="round" strokeLinejoin="round" />
          </svg>
        </div>
