import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene1_SystemReveal',
  durationInSeconds: 12,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene1_SystemReveal: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  const cameraZoom = interpolate(frame, [0, durationInFrames], [1, 1.05]);
  const word1 = sp(10);
  const word2 = sp(30);
  const word3 = sp(55);
  const lineDraw = interpolate(frame, [70, 110], [0, 1], { extrapolateRight: 'clamp' });
  const systemReveal = sp(95);
  const shake = frame > 90 && frame < 100 ? Math.sin(frame * 2) * 3 : 0;

  return (
    <AbsoluteFill style={{ backgroundColor: '#AEB784', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient overlay */}
      <AbsoluteFill style={{
        background: 'radial-gradient(circle at 30% 40%, rgba(38,38,38,0.15) 0%, transparent 60%)',
      }} />

      {/* Shape layer - large drifting rectangles */}
      <div style={{
        position: 'absolute', width: 600, height: 600, border: '2px solid #262626',
        opacity: 0.06, top: -100, right: -150,
        transform: `rotate(${frame * 0.15}deg) scale(${1 + Math.sin(frame * 0.02) * 0.05})`,
      }} />
      <div style={{
        position: 'absolute', width: 400, height: 400, borderRadius: '50%',
        background: '#262626', opacity: 0.04, bottom: -80, left: 100,
        transform: `translateY(${Math.sin(frame * 0.03) * 20}px)`,
      }} />

      {/* Motion texture - grid */}
      <svg style={{ position: 'absolute', inset: 0, opacity: 0.08 }} width="1920" height="1080">
        <defs>
          <pattern id="grid1" width="80" height="80" patternUnits="userSpaceOnUse">
            <path d="M 80 0 L 0 0 0 80" fill="none" stroke="#262626" strokeWidth="1" />
          </pattern>
        </defs>
        <rect width="1920" height="1080" fill="url(#grid1)" />
      </svg>

      {/* Floating particles */}
      {Array.from({ length: 20 }).map((_, i) => {
        const r = seededRandom(i + 1);
        const r2 = seededRandom(i + 100);
        return (
          <div key={i} style={{
            position: 'absolute',
            width: 4 + r * 6, height: 4 + r * 6, borderRadius: '50%',
            background: '#262626', opacity: 0.2,
            left: `${r * 100}%`,
            top: `${(r2 * 100 + frame * 0.1) % 100}%`,
          }} />
        );
      })}

      {/* Main content */}
      <AbsoluteFill style={{
        transform: `scale(${cameraZoom}) translateX(${shake}px)`,
        justifyContent: 'center', alignItems: 'center', flexDirection: 'column', gap: 20,
      }}>
        <div style={{
          fontFamily: 'Yantra One', fontSize: 102, color: '#262626',
          opacity: word1, transform: `translateY(${(1 - word1) * 40}px)`,
        }}>
          ये गिरावट
        </div>
        <div style={{
          fontFamily: 'Yantra One', fontSize: 102, color: '#262626',
          opacity: word2, transform: `translateY(${(1 - word2) * 40}px)`,
        }}>
          सिर्फ एक कारण नहीं
        </div>
        <div style={{
          fontFamily: 'Yantra One', fontSize: 187, color: '#FFC107',

          marginTop: 30,
        }}>
          एक पूरा तंत्र
        </div>

        {/* Underline animation */}
        <svg width="800" height="20" style={{ marginTop: 10 }}>
