import React from 'react';

import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene11_MoyennePonderee',
  durationInSeconds: 21,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene11_MoyennePonderee: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // camera slow zoom
  const camZoom = interpolate(frame, [0, durationInFrames], [1, 1.06], { extrapolateRight: 'clamp' });

  // gradient breathing
  const gradShift = Math.sin(frame * 0.02) * 8;

  // weighing scale tilt — the "moyenne pondérée" metaphor
  const scaleTilt = Math.sin(frame * 0.035) * 4;

  const products = ['TÉLÉVISEURS', 'LOYER', 'YAOURTS', 'ESSENCE', 'CAFÉ', 'SMARTPHONES'];

  const titleIn = sp(8);
  const wordReveal = (i: number) => sp(40 + i * 8);

  return (
    <AbsoluteFill style={{ backgroundColor: '#215E61', overflow: 'hidden', opacity: exitOp }}>
      {/* Dynamic gradient background */}
      <AbsoluteFill
        style={{
          background: `radial-gradient(circle at ${50 + gradShift}% ${40 - gradShift}%, #2a7a7e 0%, #215E61 45%, #163f41 100%)`,
        }}
      />

      {/* Animated shape layers — drifting transparent geometry */}
      <AbsoluteFill style={{ transform: `scale(${camZoom})` }}>
        {[0, 1, 2, 3].map((i) => {
          const r = seededRandom(i + 11);
          const drift = Math.sin(frame * 0.01 + i) * 40;
          const breathe = 1 + Math.sin(frame * 0.02 + i) * 0.06;
          return (
            <div
              key={i}
              style={{
                position: 'absolute',
                left: `${r * 80}%`,
                top: `${seededRandom(i + 21) * 70}%`,
                width: 320 + r * 220,
                height: 320 + r * 220,
                borderRadius: i % 2 === 0 ? '50%' : 24,
                border: `2px solid #FBF5A7`,
                opacity: 0.05,
                transform: `translateY(${drift}px) scale(${breathe}) rotate(${frame * 0.1 + i * 30}deg)`,
              }}
            />
          );
        })}
      </AbsoluteFill>

      {/* Motion texture — vertical grid lines */}
      <AbsoluteFill style={{ opacity: 0.06 }}>
        {Array.from({ length: 20 }).map((_, i) => (
          <div
            key={i}
            style={{
              position: 'absolute',
              left: `${(i / 20) * 100}%`,
              top: 0,
              width: 1,
              height: '100%',
              background: '#F4F2F2',
              transform: `translateX(${Math.sin(frame * 0.02 + i) * 6}px)`,
            }}
          />
        ))}
      </AbsoluteFill>

      {/* Floating particles */}
      {Array.from({ length: 24 }).map((_, i) => {
        const r = seededRandom(i + 99);
        const y = (frame * (0.4 + r) + r * 1080) % 1080;
        return (
          <div
            key={i}
            style={{
              position: 'absolute',
              left: `${r * 100}%`,
              top: y,
              width: 4 + r * 4,
              height: 4 + r * 4,
              borderRadius: '50%',
              background: '#FBF5A7',
              opacity: 0.18,
            }}
          />
        );
      })}

      {/* Main content */}
      <AbsoluteFill style={{ transform: `scale(${camZoom})` }}>
        {/* Title */}
        <div
          style={{
            position: 'absolute',
            top: 110,
            left: 90,
            opacity: titleIn,
            transform: `translateY(${(1 - titleIn) * 40}px)`,
          }}
        >
          <div style={{ fontFamily: 'Anton', fontSize: 102, color: '#F4F2F2', lineHeight: 0.95, letterSpacing: 1 }}>
            UNE MOYENNE
          </div>
          <div style={{ fontFamily: 'Anton', fontSize: 102, color: '#FBF5A7', lineHeight: 0.95, letterSpacing: 1 }}>
            PONDÉRÉE
          </div>
          <div style={{ fontFamily: 'Montserrat', fontWeight: 600, fontSize: 38, color: '#F4F2F2', opacity: 0.7, marginTop: 14 }}>
            INSEE · des centaines de produits
          </div>
        </div>

        {/* Weighing scale metaphor — central pivot */}
        <div
          style={{
            position: 'absolute',
            top: 540,
            left: '50%',
            transform: `translateX(-50%) rotate(${scaleTilt}deg)`,
            transformOrigin: 'center top',
            width: 900,
            height: 6,
            background: '#FBF5A7',
            borderRadius: 3,
            opacity: sp(60),
          }}
        >
          {/* pivot */}
          <div style={{ position: 'absolute', left: '50%', top: -18, transform: 'translateX(-50%)', width: 36, height: 36, borderRadius: '50%', background: '#F4F2F2' }} />
        </div>

        {/* Product chips floating in */}
        {products.map((p, i) => {
          const rev = wordReveal(i);
          const angle = (i / products.length) * 360;
          const orbit = 200;
          const cx = 960 + Math.cos((angle + frame * 0.3) * Math.PI / 180) * orbit;
          const cy = 760 + Math.sin((angle + frame * 0.3) * Math.PI / 180) * 70;
          return (
            <div
              key={i}
              style={{
                position: 'absolute',
                left: cx,
                top: cy,
                transform: `translate(-50%,-50%) scale(${rev})`,
                opacity: rev,
                padding: '14px 26px',
                background: i % 2 === 0 ? '#FBF5A7' : 'transparent',
                border: '2px solid #FBF5A7',
                borderRadius: 40,
                fontFamily: 'Space Mono',
                fontWeight: 700,
                fontSize: 30,
                color: i % 2 === 0 ? '#215E61' : '#F4F2F2',
                whiteSpace: 'nowrap',
              }}
            >
              {p}
            </div>
          );
        })}
      </AbsoluteFill>

      {/* Vignette overlay */}
      <AbsoluteFill
        style={{
          boxShadow: 'inset 0 0 360px 90px rgba(0,0,0,0.55)',
          pointerEvents: 'none',
        }}
      />

      {/* Watermark */}
      <div
        style={{
          position: 'absolute',
          bottom: 30,
          right: 45,
          fontFamily: 'Montserrat',
          fontWeight: 600,
          fontSize: 22,
          color: '#9E9E9E',
          opacity: interpolate(frame, [0, 30], [0, 0.3], { extrapolateRight: 'clamp' }),
        }}
      >
        Matures Finance
      </div>
    </AbsoluteFill>
  );

export default Scene11_MoyennePonderee;
