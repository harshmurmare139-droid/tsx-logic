export const compositionConfig = {
  id: 'Scene15TheTrap',
  durationInSeconds: 16,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene15TheTrap: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  const camPan = interpolate(frame, [0, durationInFrames], [10, -10], { extrapolateRight: 'clamp' });

  const steps = [
    { n: '01', t: 'SIGN UP', d: 'Employer plan' },
    { n: '02', t: 'PICK BEST', d: 'Top performing fund' },
    { n: '03', t: 'FORGET IT', d: 'Set & forget' },
  ];

  return (
    <AbsoluteFill style={{ backgroundColor: '#F9F6F0', overflow: 'hidden', opacity: exitOp, fontFamily: 'Inter, Poppins' }}>
      {/* Dynamic gradient background */}
      <AbsoluteFill style={{ background: 'linear-gradient(160deg, rgba(255,193,7,0.06), transparent 50%)' }} />

      {/* Shape layers */}
      {[0, 1, 2].map((i) => {
        const r = seededRandom(i + 55);
        return (
          <div key={i} style={{
            position: 'absolute', width: 340 + r * 220, height: 340 + r * 220,
            left: `${r * 72}%`, top: `${r * 50}%`, border: '2px solid #5A524E',
            opacity: 0.045, clipPath: i === 0 ? 'polygon(50% 0%, 0% 100%, 100% 100%)' : 'none',
            borderRadius: i === 0 ? '0' : '22px',
            transform: `rotate(${frame * 0.05 + i * 35}deg) scale(${1 + Math.sin(frame * 0.02 + i) * 0.04})`,
          }} />
        );
      })}

      {/* Motion texture: diagonal hatch */}
      <AbsoluteFill style={{ backgroundImage: 'repeating-linear-gradient(45deg, #5A524E 0 1px, transparent 1px 40px)', opacity: 0.04 }} />
