import React from 'react';

import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene01WakeUp',
  durationInSeconds: 30,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene01WakeUp: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Camera slow zoom
  const camScale = interpolate(frame, [0, durationInFrames], [1.04, 1.12], { extrapolateRight: 'clamp' });

  // Gradient drift
  const gradShift = Math.sin(frame * 0.012) * 8;

  // Title reveals
  const titleIn = sp(10);
  const sub1 = sp(70);
  const phoneIn = sp(110, { damping: 14 });

  // LinkedIn post staggered entries
  const post = (i: number) => {
    const p = sp(150 + i * 35, { damping: 16 });
    const op = interpolate(frame, [150 + i * 35, 180 + i * 35], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });
    return { y: (1 - p) * 60, op };
  };

  // Sinking feeling pulse near end
  const pulse = 1 + Math.sin(frame * 0.12) * 0.02;

  return (
    <AbsoluteFill style={{ backgroundColor: '#CFFFE2', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient + Shape Layers + Texture + Particles + Content + Vignette + Watermark */}

      {/* Dynamic gradient background */}
      <AbsoluteFill
        style={{
          background: `radial-gradient(circle at ${50 + gradShift}% ${40 - gradShift}%, #DFFFEC 0%, #CFFFE2 55%, #B6F2D2 100%)`,
        }}
      />

      {/* Animated transparent shape layers */}
      <AbsoluteFill style={{ transform: `scale(${camScale})` }}>
        <div style={{ position: 'absolute', top: '8%', left: '6%', width: 520, height: 520, borderRadius: '50%', border: '3px solid #000000', opacity: 0.05, transform: `rotate(${frame * 0.15}deg) scale(${1 + Math.sin(frame * 0.02) * 0.04})` }} />
        <div style={{ position: 'absolute', bottom: '-6%', right: '4%', width: 600, height: 600, background: '#000000', opacity: 0.04, transform: `rotate(${frame * 0.1}deg)` }} />
        <div style={{ position: 'absolute', top: '30%', right: '18%', width: 300, height: 300, borderRadius: '50%', background: '#FFC107', opacity: 0.07, transform: `scale(${1 + Math.sin(frame * 0.03) * 0.08})` }} />
      </AbsoluteFill>

      {/* Motion texture: grid lines */}
      <AbsoluteFill style={{ opacity: 0.06 }}>
        {Array.from({ length: 14 }).map((_, i) => (
          <div key={`v${i}`} style={{ position: 'absolute', left: `${i * 7.2}%`, top: 0, width: 1, height: '100%', background: '#000000', transform: `translateY(${Math.sin(frame * 0.02 + i) * 6}px)` }} />
        ))}
      </AbsoluteFill>

      {/* Floating particles */}
      {Array.from({ length: 18 }).map((_, i) => {
        const r = seededRandom(i + 1);
        const r2 = seededRandom(i + 50);
        const x = r * 100;
        const y = (r2 * 100 + frame * 0.25) % 100;
        return (
          <div key={`p${i}`} style={{ position: 'absolute', left: `${x}%`, top: `${y}%`, width: 6 + r * 8, height: 6 + r * 8, borderRadius: '50%', background: '#000000', opacity: 0.08 }} />
        );
      })}

      {/* Main content */}
      <AbsoluteFill style={{ transform: `scale(${camScale})` }}>
        {/* Left text block */}
        <div style={{ position: 'absolute', left: 110, top: 240, width: 820 }}>
          <div style={{ fontFamily: 'Mont Heavy, Inter', fontSize: 110, lineHeight: 1.0, fontWeight: 900, color: '#000000', opacity: titleIn, transform: `translateY(${(1 - titleIn) * 50}px)` }}>
            PICTURE
          </div>
          <div style={{ fontFamily: 'Mont Heavy, Inter', fontSize: 110, lineHeight: 1.0, fontWeight: 900, color: '#000000', opacity: titleIn, transform: `translateY(${(1 - titleIn) * 50}px)`, background: '#FFC107', display: 'inline-block', padding: '4px 18px', marginTop: 8 }}>
            THIS
          </div>
          <div style={{ fontFamily: 'Inter, Poppins', fontSize: 54, fontWeight: 600, color: '#000000', marginTop: 40, opacity: sub1, transform: `translateY(${(1 - sub1) * 40}px)` }}>

            <div style={{ width: 38, height: 38, borderRadius: 8, background: '#000000' }} />
