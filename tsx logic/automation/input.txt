export const compositionConfig = {
  id: 'S01_WealthShock',
  durationInSeconds: 21,
  fps: 30,
  width: 1920,
  height: 1080,
};

const C = {
  bg: '#C8A96B',
  ink: '#080616',
  paper: '#F7F4ED',
  bronze: '#8C6D3F',
  sand: '#E3D2AC',
  rust: '#B5342A',
  forest: '#1F3A34',
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

  const { fps, durationInFrames } = useVideoConfig();

  const camX = Math.sin(frame * 0.009) * 16;
  const camY = Math.cos(frame * 0.007) * 10;

  // Hero counter 0 -> 1688
  const countP = interpolate(frame, [18, 105], [0, 1], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.out(Easing.cubic),
  });
  const bigNum = Math.round(1688 * countP);

  // Impact shake when counter locks
  const shake = frame > 100 && frame < 118 ? Math.sin(frame * 2.4) * (118 - frame) * 0.7 : 0;

  // Donut chart fill 0 -> 50%
  const donutP = interpolate(frame, [150, 245], [0, 0.5], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.inOut(Easing.quad),
  });
  const R = 132;
  const CIRC = 2 * Math.PI * R;

  const wealthIn = sp(300);
  const stampIn = sp(380, { damping: 9, stiffness: 90 });

  return (
    <AbsoluteFill style={{ backgroundColor: C.bg, overflow: 'hidden', opacity: exitOp }}>
      {/* Layer 1 — dynamic gradient background */}
      <AbsoluteFill
        style={{
          background: `radial-gradient(circle at ${30 + Math.sin(frame * 0.01) * 8}% 30%, #DCC08A 0%, ${C.bg} 45%, #9C8149 100%)`,
        }}
      />

      {/* Layer 2 — animated shape layers (slow drifting geometry) */}
      <AbsoluteFill>
        <svg width="100%" height="100%" viewBox="0 0 1920 1080">
          <circle cx={340} cy={860} r={420} fill={C.ink} opacity={0.05}
            transform={`translate(0 ${Math.sin(frame * 0.012) * 26}) scale(${1 + Math.sin(frame * 0.02) * 0.04})`}
            style={{ transformOrigin: '340px 860px' }} />
          <rect x={1350} y={-120} width={620} height={620} fill={C.forest} opacity={0.06}
            transform={`rotate(${12 + frame * 0.05} 1660 190)`} />
          <polygon points="1720,980 1920,620 1920,1080" fill={C.paper} opacity={0.07}
            transform={`translate(${Math.cos(frame * 0.01) * 20} 0)`} />
        </svg>
      </AbsoluteFill>

      {/* Layer 3 — motion texture: halftone dot field (punk collage print feel) */}
      <AbsoluteFill style={{ opacity: 0.16 }}>
        <svg width="100%" height="100%">
          <defs>
            <pattern id="s01dots" width="26" height="26" patternUnits="userSpaceOnUse"
              patternTransform={`translate(${(frame * 0.35) % 26} ${(frame * 0.2) % 26})`}>
              <circle cx="6" cy="6" r="3.1" fill={C.ink} />
            </pattern>
          </defs>
          <rect width="100%" height="100%" fill="url(#s01dots)" />
        </svg>
      </AbsoluteFill>

      {/* Layer 4 — floating particles */}
      <AbsoluteFill>
        {new Array(26).fill(0).map((_, i) => {
          const sx = seededRandom(i + 1) * 1920;
          const sy = seededRandom(i + 40) * 1080;
          const sz = 3 + seededRandom(i + 90) * 7;
          const drift = Math.sin(frame * 0.02 + i) * 22;
          return (
            <div key={i} style={{
              position: 'absolute', left: sx, top: (sy + frame * 0.35) % 1080,
              width: sz, height: sz, borderRadius: '50%',
              backgroundColor: i % 3 === 0 ? C.paper : C.ink,

      {/* Layer 5 — main content with spring animations */}

        {/* Torn-paper kicker strip */}

          position: 'absolute', left: 130, top: 118,

            <circle cx="210" cy="210" r={R} fill="none" stroke={C.ink} strokeWidth="52" opacity="0.18" />
